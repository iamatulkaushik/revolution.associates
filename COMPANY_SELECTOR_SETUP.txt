# 3. Update your models to include company field
# -----------------------------------------------
# Ensure all models have a company ForeignKey:

from Sapp.app.company import Company

class Department(models.Model):
    company = models.ForeignKey(Company, on_delete=models.CASCADE, related_name='departments')
    # ... other fields

class Branch(models.Model):
    company = models.ForeignKey(Company, on_delete=models.CASCADE, related_name='branches')
    # ... other fields

class Employee(models.Model):
    company = models.ForeignKey(Company, on_delete=models.CASCADE, related_name='employees')
    # ... other fields


# 4. Use mixins in your views
# ----------------------------
# Example usage in Aapp/app/branch_department.py:

from django.views.generic import ListView, CreateView
from Aapp.mixins import CompanyFilterMixin, CompanyFormMixin

class DepartmentListView(CompanyFilterMixin, ListView):
    model = Department
    template_name = 'department_list.html'

class DepartmentCreateView(CompanyFormMixin, CreateView):
    model = Department
    fields = ['name', 'description']
    template_name = 'department_form.html'


# 5. For function-based views, filter manually
# ---------------------------------------------
# Example:

@login_required
def list_department(request):
    if not request.selected_company:
        messages.warning(request, 'Please select a company first.')
        return redirect('Aapp:dashboard')
    
    departments = Department.objects.filter(company=request.selected_company)
    return render(request, 'department_list.html', {'departments': departments})


# 6. Access selected company in templates
# ----------------------------------------
